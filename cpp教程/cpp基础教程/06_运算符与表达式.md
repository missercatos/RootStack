本章将全面介绍 C++ 的各类运算符：算术运算符、关系运算符、逻辑运算符、位运算符、
赋值运算符、自增自减运算符、三目运算符，并深入讲解运算符优先级和表达式求值

1.1 算术运算符
---------------

| 运算符 | 含义 | 示例 | 结果 |
|--------|------|------|------|
| `+` | 加法 | `7 + 3` | `10` |
| `-` | 减法 | `7 - 3` | `4` |
| `*` | 乘法 | `7 * 3` | `21` |
| `/` | 除法 | `7 / 3` | `2` |
| `%` | 取模(余数) | `7 % 3` | `1` |

```cpp
#include <iostream>

int main() {
    int a = 17, b = 5;

    std::cout << a << " + " << b << " = " << a + b << std::endl;   // 22
    std::cout << a << " - " << b << " = " << a - b << std::endl;   // 12
    std::cout << a << " * " << b << " = " << a * b << std::endl;   // 85
    std::cout << a << " / " << b << " = " << a / b << std::endl;   // 3（整数除法）
    std::cout << a << " % " << b << " = " << a % b << std::endl;   // 2

    // 整数除法 vs 浮点除法
    std::cout << "7 / 2 = " << 7 / 2 << std::endl;       // 3
    std::cout << "7.0 / 2 = " << 7.0 / 2 << std::endl;   // 3.5
    std::cout << "7 / 2.0 = " << 7 / 2.0 << std::endl;   // 3.5

    // 负数的除法和取模
    std::cout << "-7 / 2 = " << -7 / 2 << std::endl;     // -3（向零截断）
    std::cout << "-7 % 2 = " << -7 % 2 << std::endl;     // -1
    std::cout << "7 % -2 = " << 7 % -2 << std::endl;     // 1

    return 0;
}
```

1.2 关系运算符
---------------

关系运算符比较两个值，返回 bool 类型（true 或 false）：

| 运算符 | 含义 | 示例 | 结果 |
|--------|------|------|------|
| `>` | 大于 | `5 > 3` | `true` |
| `<` | 小于 | `5 < 3` | `false` |
| `>=` | 大于等于 | `5 >= 5` | `true` |
| `<=` | 小于等于 | `5 <= 3` | `false` |
| `==` | 等于 | `5 == 5` | `true` |
| `!=` | 不等于 | `5 != 3` | `true` |

```cpp
#include <iostream>

int main() {
    int x = 10, y = 20;

    std::cout << std::boolalpha;
    std::cout << "x > y: " << (x > y) << std::endl;    // false
    std::cout << "x < y: " << (x < y) << std::endl;    // true
    std::cout << "x >= 10: " << (x >= 10) << std::endl; // true
    std::cout << "x <= y: " << (x <= y) << std::endl;   // true
    std::cout << "x == y: " << (x == y) << std::endl;   // false
    std::cout << "x != y: " << (x != y) << std::endl;   // true

    // 浮点数比较的注意事项
    double a = 0.1 + 0.2;
    double b = 0.3;
    std::cout << "0.1 + 0.2 == 0.3: " << (a == b) << std::endl;  // false!

    // 正确的浮点比较
    double epsilon = 1e-9;
    std::cout << "近似相等: " << (std::abs(a - b) < epsilon) << std::endl;  // true

    return 0;
}
```

1.3 逻辑运算符
---------------

| 运算符 | 含义 | 示例 | 结果 |
|--------|------|------|------|
| `&&` | 逻辑与 | `true && false` | `false` |
| `\|\|` | 逻辑或 | `true \|\| false` | `true` |
| `!` | 逻辑非 | `!true` | `false` |

```cpp
#include <iostream>

int main() {
    int age = 25;
    double salary = 8000;
    bool hasLicense = true;

    std::cout << std::boolalpha;

    // 逻辑与：两个条件都为 true 才为 true
    bool canBuyCar = (salary > 5000) && hasLicense;
    std::cout << "能买车: " << canBuyCar << std::endl;  // true

    // 逻辑或：至少一个为 true 就为 true
    bool isWeekend = false;
    bool isHoliday = true;
    bool canRest = isWeekend || isHoliday;
    std::cout << "能休息: " << canRest << std::endl;  // true

    // 逻辑非：取反
    bool isBusy = true;
    std::cout << "不忙: " << !isBusy << std::endl;  // false

    // 组合条件
    bool eligible = (age >= 18 && age <= 60) && (salary > 3000 || hasLicense);
    std::cout << "符合条件: " << eligible << std::endl;  // true

    return 0;
}
```

2.1 位运算符
--------------

位运算符直接操作数据的二进制位：

| 运算符 | 含义 | 示例 | 结果 |
|--------|------|------|------|
| `&` | 按位与 | `5 & 3` | `1` |
| `\|` | 按位或 | `5 \| 3` | `7` |
| `^` | 按位异或 | `5 ^ 3` | `6` |
| `~` | 按位取反 | `~5` | `-6` |
| `<<` | 左移 | `5 << 1` | `10` |
| `>>` | 右移 | `5 >> 1` | `2` |

```cpp
#include <iostream>
#include <bitset>

int main() {
    int a = 0b1010;  // 10
    int b = 0b1100;  // 12

    std::cout << "a = " << std::bitset<8>(a) << " (" << a << ")" << std::endl;
    std::cout << "b = " << std::bitset<8>(b) << " (" << b << ")" << std::endl;

    // 按位与：两位都为 1 结果才为 1
    std::cout << "a & b = " << std::bitset<8>(a & b) << " (" << (a & b) << ")" << std::endl;

    // 按位或：至少一位为 1 结果就为 1
    std::cout << "a | b = " << std::bitset<8>(a | b) << " (" << (a | b) << ")" << std::endl;

    // 按位异或：两位不同结果为 1
    std::cout << "a ^ b = " << std::bitset<8>(a ^ b) << " (" << (a ^ b) << ")" << std::endl;

    // 按位取反：0 变 1，1 变 0
    std::cout << "~a = " << std::bitset<8>(~a) << " (" << (~a) << ")" << std::endl;

    // 左移：相当于乘以 2^n
    std::cout << "a << 2 = " << std::bitset<8>(a << 2) << " (" << (a << 2) << ")" << std::endl;

    // 右移：相当于除以 2^n
    std::cout << "a >> 1 = " << std::bitset<8>(a >> 1) << " (" << (a >> 1) << ")" << std::endl;

    return 0;
}
```

位运算的常用技巧：

```cpp
#include <iostream>

int main() {
    int n = 42;

    // 判断奇偶
    if (n & 1) {
        std::cout << n << " 是奇数" << std::endl;
    } else {
        std::cout << n << " 是偶数" << std::endl;
    }

    // 乘以 2 / 除以 2
    std::cout << n << " * 2 = " << (n << 1) << std::endl;
    std::cout << n << " / 2 = " << (n >> 1) << std::endl;

    // 交换两个数（不用临时变量）
    int x = 5, y = 9;
    x ^= y;
    y ^= x;
    x ^= y;
    std::cout << "交换后: x=" << x << ", y=" << y << std::endl;

    // 判断是否为 2 的幂
    int m = 64;
    bool isPowerOf2 = (m > 0) && ((m & (m - 1)) == 0);
    std::cout << m << " 是 2 的幂: " << std::boolalpha << isPowerOf2 << std::endl;

    // 获取最低位的 1
    int lowest = n & (-n);
    std::cout << n << " 最低位的 1: " << lowest << std::endl;

    return 0;
}
```

2.2 自增自减运算符（++/--）
-----------------------------

```cpp
#include <iostream>

int main() {
    int a = 5;

    // 前缀：先加/减，再使用
    std::cout << "++a = " << ++a << std::endl;   // a变为6，输出6
    std::cout << "a = " << a << std::endl;       // 6

    // 后缀：先使用，再加/减
    std::cout << "a++ = " << a++ << std::endl;   // 输出6，然后a变为7
    std::cout << "a = " << a << std::endl;       // 7

    int b = 10;
    std::cout << "--b = " << --b << std::endl;   // b变为9，输出9
    std::cout << "b-- = " << b-- << std::endl;   // 输出9，然后b变为8
    std::cout << "b = " << b << std::endl;       // 8

    // 在表达式中的区别
    int x = 3;
    int y = ++x * 2;    // x先变为4，y = 4 * 2 = 8
    std::cout << "x=" << x << " y=" << y << std::endl;

    int m = 3;
    int n = m++ * 2;    // n = 3 * 2 = 6，然后m变为4
    std::cout << "m=" << m << " n=" << n << std::endl;

    return 0;
}
```

前缀 vs 后缀的性能差异：
- 对于基本类型（int 等），编译器通常会优化，没有性能差异
- 对于迭代器和自定义类型，前缀 `++it` 比后缀 `it++` 更高效（后缀需要创建临时副本）

2.3 三目运算符（?:）
---------------------

三目运算符是唯一的三元运算符，格式为 `条件 ? 真值 : 假值`：

```cpp
#include <iostream>

int main() {
    int a = 10, b = 20;

    // 等价于 if-else 赋值
    int max = (a > b) ? a : b;
    std::cout << "较大值: " << max << std::endl;

    int min = (a < b) ? a : b;
    std::cout << "较小值: " << min << std::endl;

    // 可以嵌套使用
    int x = 15;
    std::string category = (x < 10) ? "小" : (x < 20) ? "中" : "大";
    std::cout << x << " 属于: " << category << std::endl;

    // 在输出中使用
    int score = 75;
    std::cout << "成绩: " << ((score >= 60) ? "及格" : "不及格") << std::endl;

    // 三目运算符可以作为左值（当两个分支都是左值时）
    int c = 1, d = 2;
    ((a > b) ? c : d) = 100;
    std::cout << "c=" << c << " d=" << d << std::endl;  // c=1, d=100

    return 0;
}
```

2.4 运算符优先级表
-------------------

从高到低的优先级（常用部分）：

| 优先级 | 运算符 | 结合性 |
|--------|--------|--------|
| 1 | `()` `[]` `->` `.` `++`(后缀) `--`(后缀) | 左→右 |
| 2 | `!` `~` `++`(前缀) `--`(前缀) `+`(正) `-`(负) `*`(解引用) `&`(取地址) `sizeof` | 右→左 |
| 3 | `*` `/` `%` | 左→右 |
| 4 | `+` `-` | 左→右 |
| 5 | `<<` `>>` | 左→右 |
| 6 | `<` `<=` `>` `>=` | 左→右 |
| 7 | `==` `!=` | 左→右 |
| 8 | `&` | 左→右 |
| 9 | `^` | 左→右 |
| 10 | `\|` | 左→右 |
| 11 | `&&` | 左→右 |
| 12 | `\|\|` | 左→右 |
| 13 | `?:` | 右→左 |
| 14 | `=` `+=` `-=` 等赋值 | 右→左 |
| 15 | `,` | 左→右 |

```cpp
#include <iostream>

int main() {
    // 优先级示例
    int result;

    result = 2 + 3 * 4;       // * 优先于 +，= 2 + 12 = 14
    std::cout << "2 + 3 * 4 = " << result << std::endl;

    result = (2 + 3) * 4;     // 括号最高优先级，= 5 * 4 = 20
    std::cout << "(2 + 3) * 4 = " << result << std::endl;

    // 关系和逻辑运算符
    bool b = 3 > 2 && 4 < 5;  // > 和 < 先算，再算 &&
    std::cout << "3 > 2 && 4 < 5: " << std::boolalpha << b << std::endl;

    // 位运算优先级低于比较运算符（容易出错！）
    int x = 5;
    // if (x & 1 == 0)  // 错误！== 优先于 &，实际是 x & (1 == 0) = x & 0 = 0
    if ((x & 1) == 0) {
        std::cout << "偶数" << std::endl;
    } else {
        std::cout << "奇数" << std::endl;
    }

    return 0;
}
```

2.5 表达式求值顺序
--------------------

C++ 中大多数运算符不保证操作数的求值顺序：

```cpp
#include <iostream>

int main() {
    // 短路求值：&& 和 || 保证从左到右，且可能短路
    int x = 0;
    if (x != 0 && 10 / x > 1) {
        std::cout << "条件成立" << std::endl;
    } else {
        std::cout << "短路保护，避免除零" << std::endl;
    }

    int* p = nullptr;
    if (p != nullptr && *p > 0) {
        std::cout << "安全访问" << std::endl;
    }

    // || 短路
    bool found = false;
    if (found || (found = true)) {
        std::cout << "found = " << std::boolalpha << found << std::endl;
    }

    // 未指定求值顺序的情况（避免写这样的代码）
    int i = 0;
    // int result = ++i + ++i;  // 未定义行为！不要这样写

    // 正确做法
    int a = 1, b = 1;
    ++a;
    ++b;
    int result = a + b;
    std::cout << "result = " << result << std::endl;

    return 0;
}
```

短路求值规则：
- `&&`：左边为 false 时，右边不再求值
- `||`：左边为 true 时，右边不再求值

3.1 案例一：位操作实现权限管理
-------------------------------

```cpp
#include <iostream>

const int PERM_READ    = 1 << 0;  // 0001
const int PERM_WRITE   = 1 << 1;  // 0010
const int PERM_EXECUTE = 1 << 2;  // 0100
const int PERM_DELETE  = 1 << 3;  // 1000

void showPermissions(int perm) {
    std::cout << "权限: ";
    if (perm & PERM_READ)    std::cout << "读 ";
    if (perm & PERM_WRITE)   std::cout << "写 ";
    if (perm & PERM_EXECUTE) std::cout << "执行 ";
    if (perm & PERM_DELETE)  std::cout << "删除 ";
    std::cout << std::endl;
}

int main() {
    int userPerm = 0;

    // 添加权限（使用 |）
    userPerm |= PERM_READ;
    userPerm |= PERM_WRITE;
    std::cout << "添加读写权限后:" << std::endl;
    showPermissions(userPerm);

    // 添加执行权限
    userPerm |= PERM_EXECUTE;
    std::cout << "添加执行权限后:" << std::endl;
    showPermissions(userPerm);

    // 移除写权限（使用 & ~）
    userPerm &= ~PERM_WRITE;
    std::cout << "移除写权限后:" << std::endl;
    showPermissions(userPerm);

    // 切换权限（使用 ^）
    userPerm ^= PERM_DELETE;
    std::cout << "切换删除权限后:" << std::endl;
    showPermissions(userPerm);

    // 检查是否有某权限
    bool canWrite = (userPerm & PERM_WRITE) != 0;
    std::cout << "有写权限: " << std::boolalpha << canWrite << std::endl;

    return 0;
}
```

3.2 案例二：表达式计算器
-------------------------

```cpp
#include <iostream>
#include <cmath>

int main() {
    std::cout << "=== 运算符综合演示 ===" << std::endl;

    int a, b;

    std::cout << "a - b = " << a - b << std::endl;
    std::cout << "a * b = " << a * b << std::endl;
    if (b != 0) {

        std::cout << "a % b = " << a % b << std::endl;
    } else {
        std::cout << "除数为零，无法计算除法和取模" << std::endl;
    }

    std::cout << "\n--- 关系运算 ---" << std::endl;
    std::cout << std::boolalpha;
    std::cout << "a > b: " << (a > b) << std::endl;
    std::cout << "a < b: " << (a < b) << std::endl;
    std::cout << "a == b: " << (a == b) << std::endl;
    std::cout << "a != b: " << (a != b) << std::endl;

    std::cout << "\n--- 位运算 ---" << std::endl;
    std::cout << "a & b = " << (a & b) << std::endl;
    std::cout << "a | b = " << (a | b) << std::endl;
    std::cout << "a ^ b = " << (a ^ b) << std::endl;
    std::cout << "a << 1 = " << (a << 1) << std::endl;
    std::cout << "a >> 1 = " << (a >> 1) << std::endl;

    std::cout << "\n--- 三目运算 ---" << std::endl;
    std::cout << "较大值: " << ((a > b) ? a : b) << std::endl;
    std::cout << "较小值: " << ((a < b) ? a : b) << std::endl;
    std::cout << "绝对差: " << ((a > b) ? (a - b) : (b - a)) << std::endl;

    return 0;
}
```

3.3 案例三：利用运算符优先级简化代码
-------------------------------------

```cpp
#include <iostream>

int main() {
    // 判断闰年
    int year;
    std::cout << "输入年份: ";
    std::cin >> year;

    bool isLeap = (year % 4 == 0 && year % 100 != 0) || (year % 400 == 0);
    std::cout << year << (isLeap ? " 是" : " 不是") << "闰年" << std::endl;

    // 三个数排序（使用三目运算符和比较）
    int a, b, c;
    std::cout << "输入三个整数: ";
    std::cin >> a >> b >> c;

    // 确保 a <= b <= c
    if (a > b) { int t = a; a = b; b = t; }
    if (b > c) { int t = b; b = c; c = t; }
    if (a > b) { int t = a; a = b; b = t; }

    std::cout << "从小到大: " << a << " " << b << " " << c << std::endl;

    // 使用位运算实现快速计算
    int n;
    std::cout << "输入一个正整数: ";
    std::cin >> n;

    std::cout << n << " * 8 = " << (n << 3) << std::endl;
    std::cout << n << " / 4 = " << (n >> 2) << std::endl;
    std::cout << n << " 是" << ((n & 1) ? "奇数" : "偶数") << std::endl;

    // 判断 n 是否为 2 的幂
    if (n > 0 && (n & (n - 1)) == 0) {
        std::cout << n << " 是 2 的幂" << std::endl;
    } else {
        std::cout << n << " 不是 2 的幂" << std::endl;
    }

    return 0;
}
```

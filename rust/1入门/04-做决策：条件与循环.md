** `if` ** LLVM  SSA `if`  `Phi`φ

```text
// Rust: let y = if x > 0 { 1 } else { 0 };

// LLVM IR
// entry:
//   %cmp = icmp sgt i32 %x, 0
//   br i1 %cmp, label %then, label %else
//
// then:
//   br label %merge
//
// else:
//   br label %merge
//
// merge:
//   %y = phi i32 [1, %then], [0, %else]  ← Phi 
```

### Loop Labels break 

 `break` ****loop label

```rust
// break 
let mut count = 0;
loop {
    let mut inner = 0;
    loop {
        inner += 1;
        count += 1;
        if inner >= 5 { break; }  //  loop
    }
    if count >= 20 { break; }
}

// break 
'outer: loop {
    let mut inner = 0;
    loop {
        inner += 1;
        if inner >= 5 { break 'outer; }  //  loop
    }
}
```

### for  Range

```rust
// 0..5 → [0, 1, 2, 3, 4]
for i in 0..5 {
    print!("{} ", i);  // 0 1 2 3 4
}

// 0..=5 → [0, 1, 2, 3, 4, 5]
for i in 0..=5 {
    print!("{} ", i);  // 0 1 2 3 4 5
}

// 
for i in (0..5).rev() {
    print!("{} ", i);  // 4 3 2 1 0
}

// 
for (index, value) in arr.iter().enumerate() {
    println!("[{}] = {}", index, value);
}

// 
for i in (0..20).step_by(3) {
    print!("{} ", i);  // 0 3 6 9 12 15 18
}
```

**`for` ——`IntoIterator` trait**

```rust
// for  into_iter() 
// for item in collection { ... }

// 
let mut iter = collection.into_iter();
loop {
    match iter.next() {
        Some(item) => { ... }
        None => break,
    }
}

// 
for v in vec { }       // into_iter() →  Vecv  T
for v in &vec { }      // iter() → v  &T
for v in &mut vec { }  // iter_mut() → v  &mut T
```

###  C/C++ 

```rust

let x = 1;
let description = match x {
    1 => "one",
    2 => "two",
    _ => "other",  // _

```

```c
// C: switch  fallthrough 
int x = 1;
const char* description;
switch (x) {
    case 1:
        description = "one";
        break;        //  break  fall through
    case 2:
        description = "two";
        break;
    default:
        description = "other";
        break;
}
// C  switch
```

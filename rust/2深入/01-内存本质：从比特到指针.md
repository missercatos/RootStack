### 

```rust
use std::mem;

// 
fn print_layout<T>() {
    println!(
        "Type: {}, Size: {} bytes, Align: {} bytes",
        std::any::type_name::<T>(),
        mem::size_of::<T>(),
        mem::align_of::<T>()
    );
}

fn main() {
    print_layout::<i32>();       // 4B, align 4
    print_layout::<Vec<i32>>();  // 24B, align 8 ()
    print_layout::<Box<i32>>();  // 8B, align 8
    print_layout::<&str>();      // 16B, align 8
    print_layout::<(i32, i32)>(); // 8B, align 4
    print_layout::<(i32, i64)>(); // 16B, align 8 ()
}

//  Vec 
fn main() {
    let v: Vec<i32> = vec![10, 20, 30];
    println!("Vec ptr:    {:p}", v.as_ptr());       // 
    println!("Vec len:    {}", v.len());             // 3
    println!("Vec cap:    {}", v.capacity());        //  > 3

    // 
    for i in 0..v.len() {
        println!("v[{}] = {} @ {:p}", i, v[i], &v[i]);
        // 
    }
}
```

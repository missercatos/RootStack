```text
async fn example(x: i32) -> i32 {
    let a = do_a(x).await;
    let b = do_b(a).await;

}


enum ExampleFuture {
    //  0:  do_a
    State0 { x: i32, do_a_future: DoAFuture },
    //  1: do_a  do_b
    State1 { a: i32, do_b_future: DoBFuture },
    //  2: 
    Complete,
}

impl Future for ExampleFuture {
    type Output = i32;

    fn poll(self: Pin<&mut Self>, cx: &mut Context) -> Poll<i32> {
        loop {
            match self {
                State0 { x, do_a_future } => {
                    match do_a_future.poll(cx) {
                        Poll::Ready(a) => {
                            //  1
                            *self = State1 { a, do_b_future: do_b(a) };
                            continue; // 
                        }
                        Poll::Pending => return Poll::Pending,
                    }
                }
                State1 { a, do_b_future } => {
                    match do_b_future.poll(cx) {
                        Poll::Ready(b) => {
                            return Poll::Ready(b + 1);
                        }
                        Poll::Pending => return Poll::Pending,
                    }
                }
                Complete => panic!("polled after completion"),
            }
        }
    }
}
```

```rust
use std::future::Future;
use std::pin::Pin;
use std::task::{Context, Poll};
use std::time::{Duration, Instant};

//  Future — 
struct Delay {
    when: Instant,
}

impl Future for Delay {
    type Output = ();

    fn poll(self: Pin<&mut Self>, cx: &mut Context) -> Poll<()> {
        if Instant::now() >= self.when {
            Poll::Ready(())
        } else {
            //  waker
            let waker = cx.waker().clone();
            let when = self.when;
            std::thread::spawn(move || {
                let remaining = when - Instant::now();
                std::thread::sleep(remaining);
                waker.wake(); //  executor  poll
            });
            Poll::Pending
        }
    }
}
```

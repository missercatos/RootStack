## 一、案例项目：微型PNG解码器

我们选择一个简化的PNG图像解码器作为重构案例。这是一个约600行的C项目，支持解析PNG文件头、解压IDAT数据块、重建像素。

### 1.1 为什么选择PNG解码器？

| 特性 | 分析 |
|------|------|
| 内存密集 | 大量缓冲区和内存操作 —— 受益于所有权系统 |
| 解析密集 | PNG格式解析 —— 受益于Rust类型系统 |
| 安全关键 | 恶意PNG可触发缓冲区溢出 —— 必须安全 |
| 性能敏感 | 图像解码需高效 —— 不能牺牲性能 |
| 已知算法 | zlib解压、filter重建 —— 可用Rust生态 |

```c
/* tiny_png.c */
#include "tiny_png.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

/* ─────── CRC32 ─────── */

static const uint32_t crc32_table[256] = {
    0x00000000, 0x77073096, 0xEE0E612C, 0x990951BA,
    0x076DC419, 0x706AF48F, 0xE963A535, 0x9E6495A3,
    /* ... 实际CRC表256项，此处省略以节省篇幅 */
};

static uint32_t crc32(const uint8_t* data, size_t len) {
    uint32_t crc = 0xFFFFFFFF;
    for (size_t i = 0; i < len; i++) {
        crc = crc32_table[(crc ^ data[i]) & 0xFF] ^ (crc >> 8);
    }
    return crc ^ 0xFFFFFFFF;
}

/* ─────── 内存读取器 ─────── */

typedef struct {
    const uint8_t* data;
    size_t size;
    size_t pos;
} MemReader;

static void mr_init(MemReader* mr, const uint8_t* data, size_t size) {
    mr->data = data;
    mr->size = size;
    mr->pos = 0;
}

static int mr_read_u32(MemReader* mr, uint32_t* out) {
    if (mr->pos + 4 > mr->size) return -1;
    *out = ((uint32_t)mr->data[mr->pos] << 24) |
           ((uint32_t)mr->data[mr->pos + 1] << 16) |
           ((uint32_t)mr->data[mr->pos + 2] << 8) |
           (uint32_t)mr->data[mr->pos + 3];
    mr->pos += 4;
    return 0;
}

static int mr_read_u8(MemReader* mr, uint8_t* out) {
    if (mr->pos >= mr->size) return -1;
    *out = mr->data[mr->pos++];
    return 0;
}

static int mr_read_bytes(MemReader* mr, uint8_t* buf, size_t len) {
    if (mr->pos + len > mr->size) return -1;
    memcpy(buf, mr->data + mr->pos, len);
    mr->pos += len;
    return 0;
}

static int mr_skip(MemReader* mr, size_t len) {
    if (mr->pos + len > mr->size) return -1;
    mr->pos += len;
    return 0;
}

/* ─────── 简易zlib解压（deflate解码） ─────── */

/**
 * 本实现仅支持无压缩的deflate块（BTYPE=00）
 * 完整的deflate解码器过于复杂，此处为教学示例
 */
static int inflate_simple(const uint8_t* input, size_t input_len,
                          uint8_t* output, size_t* output_len) {
    if (input_len < 2) return -1;

    /* CMF (Compression Method and Flags) */
    uint8_t cmf = input[0];
    uint8_t cm = cmf & 0x0F;
    uint8_t cinfo = (cmf >> 4) & 0x0F;
    if (cm != 8) return -1; /* 仅支持deflate */

    /* FLG (Flags) */
    uint8_t flg = input[1];
    if ((((uint16_t)cmf << 8) | flg) % 31 != 0) return -1; /* 校验 */

    size_t pos = 2;
    size_t out_pos = 0;

    /* 如果设置了FDICT标志，跳过字典ID */
    if (flg & 0x20) {
        if (pos + 4 > input_len) return -1;
        pos += 4;
    }

    /* 解压deflate块 */
    int is_final = 0;
    while (!is_final && pos < input_len) {
        if (pos + 1 > input_len) return -1;
        uint8_t header = input[pos++];
        is_final = header & 0x01;
        uint8_t btype = (header >> 1) & 0x03;

        if (btype == 0) {
            /* 无压缩块 */
            if (pos + 4 > input_len) return -1;
            uint16_t len  = input[pos] | ((uint16_t)input[pos+1] << 8);
            uint16_t nlen = input[pos+2] | ((uint16_t)input[pos+3] << 8);
            pos += 4;
            if (len != (uint16_t)(~nlen)) return -1; /* ones' complement */
            if (pos + len > input_len) return -1;
            if (out_pos + len > *output_len) return -1;
            memcpy(output + out_pos, input + pos, len);
            out_pos += len;
            pos += len;
        } else if (btype == 1 || btype == 2) {
            /* 固定/动态Huffman —— 本简易实现不支持 */
            return -2;
        } else {
            return -1;
        }
    }

    *output_len = out_pos;
    return 0;
}

/* ─────── PNG解析 ─────── */

static const uint8_t PNG_SIGNATURE[8] = {
    0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A
};

static int parse_ihdr(MemReader* mr, PngImage* img) {
    uint32_t chunk_len, chunk_crc, calc_crc;
    uint8_t type[4];
    uint8_t ihdr_data[13];

    /* 长度 */
    if (mr_read_u32(mr, &chunk_len) != 0) return PNG_ERR_FORMAT;
    if (chunk_len != 13) return PNG_ERR_FORMAT;

    /* 类型 */
    if (mr_read_bytes(mr, type, 4) != 0) return PNG_ERR_FORMAT;

    /* IHDR数据 */
    if (mr_read_bytes(mr, ihdr_data, 13) != 0) return PNG_ERR_FORMAT;

    /* 验证类型 */
    if (memcmp(type, "IHDR", 4) != 0) return PNG_ERR_FORMAT;

    /* 解析宽度和高度（big-endian） */
    img->width = ((uint32_t)ihdr_data[0] << 24) |
                 ((uint32_t)ihdr_data[1] << 16) |
                 ((uint32_t)ihdr_data[2] << 8) |
                 (uint32_t)ihdr_data[3];
    img->height = ((uint32_t)ihdr_data[4] << 24) |
                  ((uint32_t)ihdr_data[5] << 16) |
                  ((uint32_t)ihdr_data[6] << 8) |
                  (uint32_t)ihdr_data[7];
    img->bit_depth = ihdr_data[8];
    img->color_type = ihdr_data[9];

    /* 检查CRC */
    if (mr_read_u32(mr, &chunk_crc) != 0) return PNG_ERR_FORMAT;
    /* 计算CRC需要type+data，此处简化略 */

    return PNG_OK;
}

static int png_load_internal(const uint8_t* data, size_t size, PngImage* out) {
    MemReader mr;
    mr_init(&mr, data, size);

    /* 验证PNG签名 */
    uint8_t sig[8];
    if (mr_read_bytes(&mr, sig, 8) != 0) return PNG_ERR_FORMAT;
    if (memcmp(sig, PNG_SIGNATURE, 8) != 0) return PNG_ERR_FORMAT;

    memset(out, 0, sizeof(PngImage));

    /* 收集IDAT数据 */
    uint8_t* idat_data = NULL;
    size_t idat_size = 0;
    size_t idat_cap = 0;

    int first_chunk = 1;
    while (1) {
        uint32_t chunk_len;
        uint8_t type[4];

        if (mr_read_u32(&mr, &chunk_len) != 0) break;
        if (mr_read_bytes(&mr, type, 4) != 0) break;

        if (first_chunk) {
            /* 第一个块必须是IHDR */
            if (memcmp(type, "IHDR", 4) == 0) {
                /* 重新解析IHDR（因为我们正在顺序读取） */
                uint8_t ihdr[13];
                if (mr_read_bytes(&mr, ihdr, 13) != 0) goto error_cleanup;
                out->width  = ((uint32_t)ihdr[0] << 24) | ((uint32_t)ihdr[1] << 16) | ((uint32_t)ihdr[2] << 8) | (uint32_t)ihdr[3];
                out->height = ((uint32_t)ihdr[4] << 24) | ((uint32_t)ihdr[5] << 16) | ((uint32_t)ihdr[6] << 8) | (uint32_t)ihdr[7];
                out->bit_depth = ihdr[8];
                out->color_type = ihdr[9];
                /* 跳过CRC */
                mr_skip(&mr, 4);
                first_chunk = 0;
                continue;
            } else {
                goto error_cleanup;
            }
        }

        if (memcmp(type, "IDAT", 4) == 0) {
            /* 收集IDAT数据 */
            if (idat_size + chunk_len > idat_cap) {
                idat_cap = idat_size + chunk_len + 4096;
                uint8_t* new_data = (uint8_t*)realloc(idat_data, idat_cap);
                if (!new_data) goto error_cleanup;
                idat_data = new_data;
            }
            if (mr_read_bytes(&mr, idat_data + idat_size, chunk_len) != 0) goto error_cleanup;
            idat_size += chunk_len;
            /* 跳过CRC */
            mr_skip(&mr, 4);
        } else if (memcmp(type, "IEND", 4) == 0) {
            /* 图像结束 */
            mr_skip(&mr, chunk_len + 4);
            break;
        } else {
            /* 跳过其他块 */
            mr_skip(&mr, chunk_len + 4);
        }
    }

    if (out->width == 0 || out->height == 0) goto error_cleanup;

    /* 分配像素缓冲区 */
    out->channels = 4; /* 总是输出RGBA */
    size_t pixel_count = (size_t)out->width * out->height;
    size_t raw_size = pixel_count * 4;
    out->pixels = (uint8_t*)malloc(raw_size);
    if (!out->pixels) goto error_cleanup;

    /* 解压IDAT */
    size_t decompressed_size = raw_size + out->height; /* +height用于filter字节 */
    uint8_t* decompressed = (uint8_t*)malloc(decompressed_size);
    if (!decompressed) goto error_cleanup;

    memset(decompressed, 0, decompressed_size);
    if (idat_data && idat_size > 0) {
        if (inflate_simple(idat_data, idat_size, decompressed, &decompressed_size) != 0) {
            free(decompressed);
            goto error_cleanup;
        }
    }

    /* 应用PNG滤波器 */
    size_t stride = out->width * 4 + 1; /* +1 for filter byte */
    for (uint32_t y = 0; y < out->height; y++) {
        uint8_t filter = decompressed[y * stride];
        uint8_t* row = decompressed + y * stride + 1;
        uint8_t* prev = (y > 0) ? decompressed + (y - 1) * stride + 1 : NULL;

        for (uint32_t x = 0; x < out->width * 4; x++) {
            uint8_t a = (x >= 4) ? row[x - 4] : 0;
            uint8_t b = prev ? prev[x] : 0;
            uint8_t c = (x >= 4 && prev) ? prev[x - 4] : 0;

            switch (filter) {
            case 0: /* None */ break;
            case 1: row[x] += a; break; /* Sub */
            case 2: row[x] += b; break; /* Up */
            case 3: row[x] += (uint8_t)((a + b) / 2); break; /* Average */
            case 4: {
                /* Paeth */
                int p  = (int)a + (int)b - (int)c;
                int pa = abs(p - (int)a);
                int pb = abs(p - (int)b);
                int pc = abs(p - (int)c);
                uint8_t pr;
                if (pa <= pb && pa <= pc) pr = a;
                else if (pb <= pc) pr = b;
                else pr = c;
                row[x] += pr;
                break;
            }
            default: break;
            }
        }
    }

    /* 复制到输出缓冲区（跳过filter字节） */
    for (uint32_t y = 0; y < out->height; y++) {
        memcpy(out->pixels + y * out->width * 4,
               decompressed + y * stride + 1,
               out->width * 4);
    }

    free(decompressed);
    free(idat_data);
    return PNG_OK;

error_cleanup:
    free(idat_data);
    free(out->pixels);
    out->pixels = NULL;
    return PNG_ERR_FORMAT;
}

/* ─────── 公共API ─────── */

int png_load(const char* filepath, PngImage* out) {
    if (!filepath || !out) return PNG_ERR_FILE;

    FILE* file = fopen(filepath, "rb");
    if (!file) return PNG_ERR_FILE;

    fseek(file, 0, SEEK_END);
    long size = ftell(file);
    fseek(file, 0, SEEK_SET);

    if (size <= 0) {
        fclose(file);
        return PNG_ERR_FORMAT;
    }

    uint8_t* data = (uint8_t*)malloc((size_t)size);
    if (!data) {
        fclose(file);
        return PNG_ERR_ALLOC;
    }

    size_t read = fread(data, 1, (size_t)size, file);
    fclose(file);

    if (read != (size_t)size) {
        free(data);
        return PNG_ERR_FILE;
    }

    int result = png_load_internal(data, (size_t)size, out);
    free(data);
    return result;
}

int png_load_from_memory(const uint8_t* data, size_t size, PngImage* out) {
    if (!data || !out) return PNG_ERR_FILE;
    return png_load_internal(data, size, out);
}

void png_free(PngImage* img) {
    if (img) {
        free(img->pixels);
        img->pixels = NULL;
    }
}

const char* png_error_string(int error_code) {
    switch (error_code) {
    case PNG_OK:              return "Success";
    case PNG_ERR_FILE:        return "File error";
    case PNG_ERR_FORMAT:      return "Invalid PNG format";
    case PNG_ERR_ALLOC:       return "Memory allocation failed";
    case PNG_ERR_CHECKSUM:    return "Checksum mismatch";
    case PNG_ERR_NOT_SUPPORTED: return "Feature not supported";
    default:                  return "Unknown error";
    }
}
```

## 三、C代码的安全审计

### 3.1 不安全点汇总

```c
/* 按严重性排序 */
// 致命（CRITICAL）:
// 1. 无整数溢出检查
size_t raw_size = (size_t)out->width * out->height;  // 可能溢出！
// 2. realloc 使用后未检查指针是否变化
uint8_t* new_data = (uint8_t*)realloc(idat_data, idat_cap);
idat_data = new_data;  // 如果realloc失败，idat_data悬空！

// 高危（HIGH）:
// 3. 数组越界访问
row[x] += a;  // a来自row[x-4]，需要x>=4的检查

// 4. 空指针解引用风险
if (!filepath || !out)  // 检查了但很脆弱

// 5. 文件读取大小溢出
long size = ftell(file);  // long→size_t可能有符号问题

// 中危（MEDIUM）:
// 6. goto错误处理
goto error_cleanup;  // 3个goto标签，清理路径重复

// 7. 魔法数字
uint8_t ihdr[13];  // 为什么是13？没有符号常量化

// 8. 边界检查分散
if (mr->pos + 4 > mr->size)  // 每个read函数中重复
```

```rust
    FormatError(String),
    AllocError,

            PngError::FileError(s) => write!(f, "File error: {}", s),
            PngError::FormatError(s) => write!(f, "Invalid format: {}", s),
            PngError::AllocError => write!(f, "Memory allocation failed"),
            PngError::ChecksumError => write!(f, "Checksum mismatch"),
            PngError::NotSupported(s) => write!(f, "Not supported: {}", s),
        }
    }
}

/// 安全加载PNG文件
pub fn load_png(path: &str) -> Result<PngImage, PngError> {
    let c_path = CString::new(path)
        .map_err(|e| PngError::FileError(format!("Invalid path: {}", e)))?;

    let mut raw = PngImageFFI {
        width: 0,
        height: 0,
        bit_depth: 0,
        color_type: 0,
        channels: 0,
        pixels: ptr::null_mut(),
    };

    let result = unsafe { png_load(c_path.as_ptr(), &mut raw) };

    if result != 0 {
        let err_str = unsafe { CStr::from_ptr(png_error_string(result)) };
        return Err(PngError::FormatError(
            err_str.to_string_lossy().into_owned()
        ));
    }

    // 安全转换：将C拥有的内存转移到Rust的Vec
    let pixel_count = (raw.width as usize) * (raw.height as usize) * 4;
    let pixels = unsafe {
        Vec::from_raw_parts(raw.pixels, pixel_count, pixel_count)
    };

    Ok(PngImage {
        width: raw.width,
        height: raw.height,
        bit_depth: raw.bit_depth,
        color_type: raw.color_type,
        pixels,
    })
    // raw.pixels 现在由Rust Vec拥有
    // 不要调用 png_free！Rust的Drop会处理
}

        width: 0, height: 0, bit_depth: 0, color_type: 0,

        png_load_from_memory(data.as_ptr(), data.len(), &mut raw)

        Vec::from_raw_parts(raw.pixels, pixel_count, pixel_count)
    };

    Ok(PngImage {
        width: raw.width,
        height: raw.height,
        bit_depth: raw.bit_depth,
        color_type: raw.color_type,

    })
```

## 五、策略2：核心逻辑完全重写

将`inflate_simple`和`png_load_internal`用纯安全Rust重写。

### 5.1 MemReader → Rust

```rust
/// 内存读取器 —— 替代C的MemReader，消除所有的越界读风险
struct MemReader<'a> {
    data: &'a [u8],
    pos: usize,
}

impl<'a> MemReader<'a> {
    fn new(data: &'a [u8]) -> Self {
        MemReader { data, pos: 0 }
    }

    fn pos(&self) -> usize { self.pos }

    fn remaining(&self) -> usize {
        self.data.len().saturating_sub(self.pos)
    }

    fn read_u32(&mut self) -> Option<u32> {
        if self.remaining() < 4 { return None; }
        let bytes: [u8; 4] = self.data[self.pos..self.pos + 4]
            .try_into().unwrap();
        self.pos += 4;
        Some(u32::from_be_bytes(bytes))
    }

    fn read_u8(&mut self) -> Option<u8> {
        if self.remaining() < 1 { return None; }
        let b = self.data[self.pos];
        self.pos += 1;
        Some(b)
    }

    fn read_bytes(&mut self, len: usize) -> Option<&'a [u8]> {
        if self.remaining() < len { return None; }
        let slice = &self.data[self.pos..self.pos + len];
        self.pos += len;
        Some(slice)
    }

    fn skip(&mut self, len: usize) -> Option<()> {
        if self.remaining() < len { return None; }
        self.pos += len;
        Some(())
    }
}
```

### 5.2 PNG头部解析 → Rust

```rust
const PNG_SIGNATURE: [u8; 8] = [0x89, 0x50, 0x4E, 0x47, 0x0D, 0x0A, 0x1A, 0x0A];

#[derive(Debug, Clone, Copy, PartialEq, Eq)]
enum ColorType {
    Grayscale = 0,
    Rgb = 2,
    Palette = 3,
    GrayscaleAlpha = 4,
    Rgba = 6,
}

impl ColorType {
    fn from_u8(v: u8) -> Option<ColorType> {
        match v {
            0 => Some(ColorType::Grayscale),
            2 => Some(ColorType::Rgb),
            3 => Some(ColorType::Palette),
            4 => Some(ColorType::GrayscaleAlpha),
            6 => Some(ColorType::Rgba),
            _ => None,
        }
    }

    fn channels(self) -> usize {
        match self {
            ColorType::Grayscale => 1,
            ColorType::Rgb => 3,
            ColorType::Palette => 1,
            ColorType::GrayscaleAlpha => 2,
            ColorType::Rgba => 4,
        }
    }
}

#[derive(Debug)]
struct ImageHeader {
    width: u32,
    height: u32,
    bit_depth: u8,
    color_type: ColorType,
}

fn parse_ihdr(reader: &mut MemReader) -> Result<ImageHeader, PngError> {
    let chunk_len = reader.read_u32().ok_or(PngError::FormatError("truncated IHDR length".into()))?;
    if chunk_len != 13 {
        return Err(PngError::FormatError("IHDR must be 13 bytes".into()));
    }

    let chunk_type = reader.read_bytes(4).ok_or(PngError::FormatError("truncated IHDR type".into()))?;
    if chunk_type != b"IHDR" {
        return Err(PngError::FormatError("expected IHDR chunk".into()));
    }

    let data = reader.read_bytes(13).ok_or(PngError::FormatError("truncated IHDR data".into()))?;

    let width  = u32::from_be_bytes([data[0], data[1], data[2], data[3]]);
    let height = u32::from_be_bytes([data[4], data[5], data[6], data[7]]);
    let bit_depth = data[8];
    let color_type = ColorType::from_u8(data[9])
        .ok_or_else(|| PngError::FormatError(format!("unknown color type: {}", data[9])))?;

    // 跳过CRC（4字节）
    reader.skip(4);

    if width == 0 || height == 0 {
        return Err(PngError::FormatError("zero dimension".into()));
    }

    // 检查乘积溢出
    width.checked_mul(height)
        .ok_or(PngError::FormatError("image too large".into()))?;

    Ok(ImageHeader { width, height, bit_depth, color_type })
}
```

### 5.3 PNG滤波器 → Rust

```rust
/// 应用PNG滤波器（Sub、Up、Average、Paeth）重建原始像素
fn apply_filters(
    data: &mut [u8],
    width: u32,
    height: u32,
    channels: usize,
) {
    let stride = width as usize * channels + 1; // +1 for filter byte

    for y in 0..height as usize {
        let filter = data[y * stride];
        // 当前行的数据起始位置（跳过filter byte）
        let row_start = y * stride + 1;
        let prev_start = if y > 0 { (y - 1) * stride + 1 } else { row_start + stride }; // sentinel

        for x in 0..(width as usize * channels) {
            let a = if x >= channels {
                data[row_start + x - channels]
            } else {
                0
            };
            let b = if y > 0 {
                data[prev_start + x]
            } else {
                0
            };
            let c = if x >= channels && y > 0 {
                data[prev_start + x - channels]
            } else {
                0
            };

            let idx = row_start + x;
            match filter {
                0 => {} // None
                1 => { data[idx] = data[idx].wrapping_add(a); } // Sub
                2 => { data[idx] = data[idx].wrapping_add(b); } // Up
                3 => { // Average
                    let avg = (a as u16 + b as u16) / 2;
                    data[idx] = data[idx].wrapping_add(avg as u8);
                }
                4 => { // Paeth
                    let p = a as i32 + b as i32 - c as i32;
                    let pa = (p - a as i32).abs();
                    let pb = (p - b as i32).abs();
                    let pc = (p - c as i32).abs();
                    let pr = if pa <= pb && pa <= pc { a }
                             else if pb <= pc { b }
                             else { c };
                    data[idx] = data[idx].wrapping_add(pr);
                }
                _ => {} // 未定义滤波器，保留原值
            }
        }
    }
}
```

```rust
/// 从内存中的PNG数据加载图像（纯安全Rust实现）
pub fn load_png_rust(data: &[u8]) -> Result<PngImage, PngError> {
    let mut reader = MemReader::new(data);

    // 1. 验证PNG签名
    let sig = reader.read_bytes(8)
        .ok_or(PngError::FormatError("file too small".into()))?;
    if sig != PNG_SIGNATURE {
        return Err(PngError::FormatError("not a PNG file".into()));
    }

    // 2. 解析IHDR
    let header = parse_ihdr(&mut reader)?;

    let channels = 4; // 统一输出RGBA
    let pixel_count = (header.width as usize)
        .checked_mul(header.height as usize)
        .ok_or(PngError::FormatError("image dimensions overflow".into()))?;
    let raw_size = pixel_count
        .checked_mul(channels)
        .ok_or(PngError::FormatError("pixel buffer overflow".into()))?;

    // 3. 收集IDAT数据
    let mut idat_data = Vec::new();

    loop {
        let chunk_len = match reader.read_u32() {
            Some(l) => l as usize,
            None => break,
        };

        let chunk_type = match reader.read_bytes(4) {
            Some(t) => t,
            None => break,
        };

        match chunk_type {
            b"IDAT" => {
                let chunk_data = reader.read_bytes(chunk_len)
                    .ok_or(PngError::FormatError("truncated IDAT".into()))?;
                idat_data.extend_from_slice(chunk_data);
                reader.skip(4); // CRC
            }
            b"IEND" => {
                reader.skip(chunk_len + 4);
                break;
            }
            _ => {
                // 跳过未知/辅助块
                reader.skip(chunk_len + 4);
            }
        }
    }

    // 4. 分配输出缓冲区
    let mut pixels = vec![0u8; raw_size];

    // 5. 解压IDAT（调用Rust生态的flate2或miniz_oxide）
    let decompressed = inflate_zlib(&idat_data)
        .map_err(|e| PngError::FormatError(format!("decompression failed: {}", e)))?;

    // 6. 应用滤波器
    let out_rows = header.height as usize;
    let out_stride = header.width as usize * channels + 1;

    // 复制解压数据到带滤波器字节的缓冲区
    let mut filtered = vec![0u8; out_rows * out_stride];
    for y in 0..out_rows {
        let src_start = y * (header.width as usize * channels + 1);
        let dst_start = y * out_stride;
        let row_len = (header.width as usize * channels + 1)
            .min(decompressed.len().saturating_sub(src_start));
        filtered[dst_start..dst_start + row_len]
            .copy_from_slice(&decompressed[src_start..src_start + row_len]);
    }

    apply_filters(&mut filtered, header.width, header.height, channels);

    // 7. 移除滤波器字节，复制到像素缓冲区
    for y in 0..out_rows {
        let src = &filtered[y * out_stride + 1..y * out_stride + 1 + header.width as usize * channels];
        let dst_start = y * header.width as usize * channels;
        pixels[dst_start..dst_start + src.len()].copy_from_slice(src);
    }

    Ok(PngImage {
        width: header.width,
        height: header.height,
        bit_depth: header.bit_depth,
        color_type: header.color_type as u8,
        pixels,
    })
}
```

### 一、概念题（40分，每题8分）

1. 描述将C图像解码器重构为Rust时的三种策略（FFI包装、部分重写、完全重写），分别适用于什么场景？
2. C代码中`realloc`的使用有哪些常见危险？Rust如何处理动态数组扩容？
3. 为什么PNG解码器中的滤波器（Sub/Up/Average/Paeth）是重构的高价值目标？从安全角度分析。
4. 在Rust FFI包装层中，如何安全地将C分配的内存转移给Rust的`Vec<u8>`？
5. 描述`checked_mul`在防止整数溢出方面的作用，以及C代码中缺少此类检查的后果。

<details>
<summary>参考答案</summary>

1. 三种策略：
   - FFI包装：适合C代码复杂、稳定且经过充分测试的场景，用安全Rust包装不安全C接口
   - 部分重写（混合）：保持C ABI作为外部接口，核心模块用Rust重写，适合大型项目中逐步迁移
   - 完全重写：适合涉及大量内存操作或解析的模块，将所有C代码替换为安全Rust

2. realloc常见危险：
   - 如果realloc返回NULL，原始指针可能丢失（在某些实现中原始内存仍有效）
   - 大小计算溢出导致分配过小的缓冲区
   - 使用realloc后未更新指针（如果使用了旧的指针变量）
   Rust的Vec自动处理扩容，使用安全的增长策略，内部使用checked_mul防止溢出，且不会丢失数据。

3. 滤波器的重构价值：
   - 直接操作原始像素缓冲区，涉及大量索引运算和边界计算
   - 每个像素的a/b/c值依赖于之前计算的像素，索引错误可能级联
   - C代码中边界检查分散且不完整（如x>=4检查不够）
   - 使用wrap_add避免溢出但需要确保语义正确性

4. 安全转移C内存到Vec：
   - 使用`Vec::from_raw_parts(ptr, length, capacity)`将C的malloc分配的原始指针转移到Rust所有权下
   - 必须确保length和capacity正确（等于或大于实际数据大小）
   - 之后不能再调用C的free，由Rust的Drop处理
   - 一旦转移完成，封装在safe Rust接口内，调用者无需担心

5. checked_mul的作用：
   - 返回Option，当乘法溢出时返回None
   - 在图像处理中，width*height*4可能轻易溢出32位范围
   - C代码中缺少检查的后果：分配远小于需要的缓冲区，后续写入导致堆缓冲区溢出，可能导致任意代码执行（CVE级别的安全漏洞）

潜在问题：

```rust
        .checked_mul(4)
        .ok_or("buffer size overflow")?;
    if buffer_size > isize::MAX as usize {
        return Err("buffer too large for allocation".into());
    }
    let mut buf = Vec::with_capacity(buffer_size);
    buf.resize(buffer_size, 0);
    Ok(buf)
}
```

## 本章小结

本章通过一个小型PNG解码器的重构案例，展示了三种C→Rust迁移策略的实际应用：

1. **FFI包装层**：最快的安全化路径，用~80行Rust代码包装C库，消除最危险的空指针和内存泄漏
2. **模块化重写**：从MemReader到滤波器，每个模块可独立测试和替换
3. **全面重写**：使用Rust生态（flate2/miniz_oxide）替代手写deflate，获得更好的性能和安全性

关键收获：
- 图像解码器等内存密集型代码是Rust重写的最高价值目标
- 整数溢出检查（`checked_mul`）可以预防CVE级别的漏洞
- Rust的迭代器和切片操作天然防止缓冲区溢出
- 性能不会因安全性而降低（实际上可能因更好的优化而更快）

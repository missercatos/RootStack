struct MemReader<'a> { data: &'a [u8], pos: usize }
impl<'a> MemReader<'a> {
 fn remaining(&self) -> usize { self.data.len().saturating_sub(self.pos) }
 fn read_u32(&mut self) -> Option<u32> {
 if self.remaining() < 4 { return None; }
 let bytes: [u8; 4] = self.data[self.pos..self.pos+4].try_into().unwrap();
 self.pos += 4;
 Some(u32::from_be_bytes(bytes))
 }
}

fn apply_filters(data: &mut [u8], width: u32, height: u32, channels: usize) {
 let stride = width as usize * channels + 1;
 for y in 0..height as usize {
 let filter = data[y * stride];
 let row_start = y * stride + 1;
 let prev_start = if y > 0 { (y - 1) * stride + 1 } else { row_start + stride };
 for x in 0..(width as usize * channels) {
 let a = if x >= channels { data[row_start + x - channels] } else { 0 };
 let b = if y > 0 { data[prev_start + x] } else { 0 };
 let c = if x >= channels && y > 0 { data[prev_start + x - channels] } else { 0 };
 let idx = row_start + x;
 match filter {
 1 => data[idx] = data[idx].wrapping_add(a),
 2 => data[idx] = data[idx].wrapping_add(b),
 3 => { data[idx] = data[idx].wrapping_add(((a as u16 + b as u16) / 2) as u8); }
 4 => {
 let p = a as i32 + b as i32 - c as i32;
 let pa = (p - a as i32).unsigned_abs();
 let pb = (p - b as i32).unsigned_abs();
 let pc = (p - c as i32).unsigned_abs();
 let pr = if pa <= pb && pa <= pc { a } else if pb <= pc { b } else { c };
 data[idx] = data[idx].wrapping_add(pr);
 }
 _ => {}
 }
 }
 }
}

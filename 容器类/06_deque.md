## ==========================================================================
STL 容器速通 — deque (双端队列)
## ==========================================================================

deque 支持在头部和尾部 O(1) 插入删除，底层由分段连续内存块构成，是 queue 和 stack 的默认底层容器。

## --------------------------------------------------------------------------
## 一、适用场景
## --------------------------------------------------------------------------

| 场景 | 说明 |
|------|------|
| 0-1 BFS | 权值为 0 的边 push_front，权值为 1 的边 push_back |
| 滑动窗口最值 | 维护单调双端队列，O(n) 求区间最值 |
| 需要两端插入删除 | 比 vector 少了 "只能在尾部" 的限制 |
| DFS 搜索栈 | 可当栈也可当队列使用，但一般直接选 queue/stack |

## --------------------------------------------------------------------------
## 二、声明与初始化
## --------------------------------------------------------------------------

```cpp
#include <deque>
using namespace std;

deque<int> dq;                       // 空 deque
deque<int> dq2(5, 10);               // 5 个元素，值全 10
deque<int> dq3 = {1, 2, 3};          // 列表初始化
deque<int> dq4(dq3);                 // 拷贝构造
deque<int> dq5(dq3.begin(), dq3.end()); // 迭代器范围
```

### P1886 滑动窗口 / 单调队列

题目: n 个数的序列，窗口长度为 k，求每个窗口的最小值与最大值。

```cpp
#include <iostream>
#include <deque>
using namespace std;

int a[1000005];

int main() {
    int n, k;
    cin >> n >> k;
    for (int i = 1; i <= n; i++) cin >> a[i];

    deque<int> dq; // 存下标
    // 最小值
    for (int i = 1; i <= n; i++) {
        while (!dq.empty() && a[dq.back()] >= a[i]) dq.pop_back();
        dq.push_back(i);
        if (dq.front() <= i - k) dq.pop_front();
        if (i >= k) cout << a[dq.front()] << " ";
    }
    cout << endl;

    dq.clear();
    // 最大值
    for (int i = 1; i <= n; i++) {
        while (!dq.empty() && a[dq.back()] <= a[i]) dq.pop_back();
        dq.push_back(i);
        if (dq.front() <= i - k) dq.pop_front();
        if (i >= k) cout << a[dq.front()] << " ";
    }
    return 0;
}
```

### P1346 电车（0-1 BFS）

```cpp
#include <iostream>
#include <deque>
#include <vector>
using namespace std;

const int INF = 1e9;
int dist[105];
vector<pair<int,int>> g[105];

int main() {
    int n, a, b;
    cin >> n >> a >> b;
    for (int i = 1; i <= n; i++) dist[i] = INF;
    for (int u = 1; u <= n; u++) {
        int k, v;
        cin >> k;
        for (int j = 0; j < k; j++) {
            cin >> v;
            g[u].push_back({v, j == 0 ? 0 : 1});
        }
    }
    deque<int> dq;
    dist[a] = 0;
    dq.push_front(a);
    while (!dq.empty()) {
        int u = dq.front(); dq.pop_front();
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (w == 0) dq.push_front(v);
                else dq.push_back(v);
            }
        }
    }
    cout << (dist[b] == INF ? -1 : dist[b]) << endl;
    return 0;
}
```

### P1714 切蛋糕（前缀和 + 单调队列）

```cpp
// 长度为 n 的序列，求连续不超过 m 个数的最大和
#include <iostream>
#include <deque>
using namespace std;

int sum[500005]; // 前缀和

int main() {
    int n, m, ans = -2e9;
    cin >> n >> m;
    for (int i = 1, x; i <= n; i++) {
        cin >> x;
        sum[i] = sum[i - 1] + x;
    }
    deque<int> dq; dq.push_back(0);
    for (int i = 1; i <= n; i++) {
        while (!dq.empty() && dq.front() < i - m) dq.pop_front();
        ans = max(ans, sum[i] - sum[dq.front()]);
        while (!dq.empty() && sum[dq.back()] >= sum[i]) dq.pop_back();
        dq.push_back(i);
    }
    cout << ans << endl;
    return 0;
}
```

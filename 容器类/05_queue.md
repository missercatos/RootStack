## ==========================================================================
STL 容器速通 — queue (队列)
## ==========================================================================

queue 是先进先出 (FIFO) 的容器适配器，底层默认基于 deque，限制一端插入另一端删除。

## --------------------------------------------------------------------------
## 一、适用场景
## --------------------------------------------------------------------------

| 场景 | 说明 |
|------|------|
| BFS 广度优先搜索 | 按层次逐层扩展 |
| 消息队列 | 先到先处理 |
| 生产者消费者 | 缓冲任务 |
| 模拟排队场景 | 按到达顺序处理 |
| 轮转调度 | Round Robin |

## --------------------------------------------------------------------------
## 二、声明与初始化
## --------------------------------------------------------------------------

```cpp
#include <queue>
using namespace std;

queue<int> q;                       // 默认基于 deque
queue<int, list<int>> q2;           // 基于 list（指定底层容器）
queue<int> q3(q);                   // 拷贝构造
```

## --------------------------------------------------------------------------
## 三、成员函数总览
## --------------------------------------------------------------------------

| 函数 | 说明 |
|------|------|
| `q.push(x)` | 尾部插入元素 |
| `q.emplace(args...)` | 尾部原位构造 (C++11) |
| `q.pop()` | 弹出队首元素（无返回值！） |
| `q.front()` | 返回队首元素的引用 |
| `q.back()` | 返回队尾元素的引用 |
| `q.size()` | 返回元素个数 |
| `q.empty()` | 是否为空 |
| `q.swap(q2)` | 交换两个队列的内容 |

**注意**: queue 同样没有迭代器，不能遍历。

### P1443 马的遍历（BFS）

```cpp
#include <iostream>
#include <queue>
#include <cstring>
#include <cstdio>
using namespace std;

int dist[405][405];
int dx[] = {-2,-2,-1,-1,1,1,2,2};
int dy[] = {-1,1,-2,2,-2,2,-1,1};

int main() {
    memset(dist, -1, sizeof(dist));
    int n, m, sx, sy;
    cin >> n >> m >> sx >> sy;
    queue<pair<int,int>> q;
    dist[sx][sy] = 0;
    q.push({sx, sy});
    while (!q.empty()) {
        auto [x, y] = q.front(); q.pop();
        for (int i = 0; i < 8; i++) {
            int nx = x + dx[i], ny = y + dy[i];
            if (nx>=1 && nx<=n && ny>=1 && ny<=m && dist[nx][ny]==-1) {
                dist[nx][ny] = dist[x][y] + 1;
                q.push({nx, ny});
            }
        }
    }
    for (int i = 1; i <= n; i++) {
        for (int j = 1; j <= m; j++)
            printf("%-5d", dist[i][j]);
        printf("\n");
    }
    return 0;
}
```

### P1996 约瑟夫问题

```cpp
#include <iostream>
#include <queue>
using namespace std;

int main() {
    int n, m;
    cin >> n >> m;
    queue<int> q;
    for (int i = 1; i <= n; i++) q.push(i);
    while (!q.empty()) {
        for (int i = 1; i < m; i++) {
            q.push(q.front());
            q.pop();
        }
        cout << q.front() << " ";
        q.pop();
    }
    return 0;
}
```

### P1540 [NOIP2010 提高组] 机器翻译

```cpp
#include <iostream>
#include <queue>
using namespace std;

bool inCache[1005];

int main() {
    int m, n, ans = 0;
    cin >> m >> n;
    queue<int> q;
    for (int i = 0, x; i < n; i++) {
        cin >> x;
        if (!inCache[x]) {
            ans++;
            if ((int)q.size() == m) {
                inCache[q.front()] = false;
                q.pop();
            }
            q.push(x);
            inCache[x] = true;
        }
    }
    cout << ans << endl;
    return 0;
}
```

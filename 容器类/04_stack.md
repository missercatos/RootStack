## ==========================================================================
STL 容器速通 — stack (栈)
## ==========================================================================

stack 是后进先出 (LIFO) 的容器适配器，底层默认基于 deque，限制只在一端操作。

## --------------------------------------------------------------------------
## 一、适用场景
## --------------------------------------------------------------------------

| 场景 | 说明 |
|------|------|
| 括号匹配 | 遇左括号压栈，遇右括号弹栈匹配 |
| 表达式求值 | 操作数栈 + 运算符栈 |
| DFS 非递归实现 | 手动维护搜索栈 |
| 单调栈 | 求下一个更大/更小元素 |
| 撤销操作 | 记录操作历史，撤销时弹栈 |

## --------------------------------------------------------------------------
## 二、声明与初始化
## --------------------------------------------------------------------------

```cpp
#include <stack>
using namespace std;

stack<int> st;                      // 默认基于 deque
stack<int, vector<int>> st2;        // 基于 vector（指定底层容器）
stack<int> st3(st);                 // 拷贝构造
```

## --------------------------------------------------------------------------
## 三、成员函数总览
## --------------------------------------------------------------------------

| 函数 | 说明 |
|------|------|
| `st.push(x)` | 压入元素（调用底层 `push_back`） |
| `st.emplace(args...)` | 原位构造并压入 (C++11) |
| `st.pop()` | 弹出栈顶元素（无返回值！） |
| `st.top()` | 返回栈顶元素的引用 |
| `st.size()` | 返回元素个数 |
| `st.empty()` | 是否为空 |
| `st.swap(st2)` | 交换两个栈的内容 |

**注意**: stack 没有迭代器，不能遍历。要遍历只能反复 pop+top。

## 四、洛谷实战

### P1739 表达式括号匹配

```cpp
#include <iostream>
#include <string>
#include <stack>
using namespace std;

int main() {
    string s;
    getline(cin, s);
    stack<char> st;
    for (char c : s) {
        if (c == '(') st.push(c);
        else if (c == ')') {
            if (st.empty()) { cout << "NO" << endl; return 0; }
            st.pop();
        }
    }
    cout << (st.empty() ? "YES" : "NO") << endl;
    return 0;
}
```

### P1449 后缀表达式

```cpp
#include <iostream>
#include <string>
#include <stack>
using namespace std;

int main() {
    string s;
    cin >> s;
    stack<int> st;
    int cur = 0;
    for (char c : s) {
        if (c >= '0' && c <= '9') {
            cur = cur * 10 + (c - '0');
        } else if (c == '.') {
            st.push(cur);
            cur = 0;
        } else if (c != '@') {
            int b = st.top(); st.pop();
            int a = st.top(); st.pop();
            if (c == '+') st.push(a + b);
            else if (c == '-') st.push(a - b);
            else if (c == '*') st.push(a * b);
            else if (c == '/') st.push(a / b);
        }
    }
    cout << st.top() << endl;
    return 0;
}
```

### P1981 表达式求值（中缀转后缀）

```cpp
// 只含 + 和 * 的中缀表达式求值，利用栈分优先级
#include <iostream>
#include <string>
#include <stack>
using namespace std;
const int MOD = 10000;

int main() {
    string s;
    cin >> s;
    stack<int> st;
    char op = '+';
    int cur = 0;
    for (char c : s) {
        if (c >= '0' && c <= '9') {
            cur = cur * 10 + (c - '0');
        } else {
            if (op == '*') { int t = st.top(); st.pop(); st.push(t * cur % MOD); }
            else st.push(cur % MOD);
            op = c;
            cur = 0;
        }
    }
    if (op == '*') { int t = st.top(); st.pop(); st.push(t * cur % MOD); }
    else st.push(cur % MOD);
    int ans = 0;
    while (!st.empty()) { ans = (ans + st.top()) % MOD; st.pop(); }
    cout << ans << endl;
    return 0;
}
```

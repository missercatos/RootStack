### P1346 电车
## --------------------------------------------------------------------------

题目: 城市有N个路口和若干有向电车轨道。每条轨道有默认方向（无需切换）和反向（需要切换）。
求从起点到终点的最少切换次数。

思路: 将切换次数视为边权（默认方向权0，切换方向权1），跑0-1 BFS或Dijkstra。

```cpp
#include <iostream>
#include <queue>
#include <vector>
using namespace std;

const int INF = 1e9;
int n, a, b, dist[105];
vector<pair<int,int>> g[105];

int main() {
    cin >> n >> a >> b;
    for (int i = 1; i <= n; i++) dist[i] = INF;
    for (int u = 1; u <= n; u++) {
        int k, v;
        cin >> k;
        for (int j = 0; j < k; j++) {
            cin >> v;
            g[u].push_back({v, j == 0 ? 0 : 1});
        }
    }
    deque<int> q;
    dist[a] = 0;
    q.push_front(a);
    while (!q.empty()) {
        int u = q.front(); q.pop_front();
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (w == 0) q.push_front(v);
                else q.push_back(v);
            }
        }
    }
    cout << (dist[b] == INF ? -1 : dist[b]) << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P1119 灾后重建
## --------------------------------------------------------------------------

题目: N个村庄，部分村庄间有道路连接，道路在第t_i天后修好。有Q个询问(x,y,t)，
询问第t天时x到y的最短路径。

思路: Floyd算法的优化用法，按道路修好的时间逐个加入中间节点进行松弛。

```cpp
#include <iostream>
using namespace std;

const int INF = 0x3f3f3f3f;
int n, m, t[205], f[205][205];

int main() {
    cin >> n >> m;
    for (int i = 0; i < n; i++) cin >> t[i];
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n; j++)
            f[i][j] = (i == j ? 0 : INF);
    for (int i = 0, u, v, w; i < m; i++) {
        cin >> u >> v >> w;
        f[u][v] = f[v][u] = w;
    }
    int Q, cur = 0;
    cin >> Q;
    while (Q--) {
        int x, y, d;
        cin >> x >> y >> d;
        while (cur < n && t[cur] <= d) {
            for (int i = 0; i < n; i++)
                for (int j = 0; j < n; j++)
                    if (f[i][j] > f[i][cur] + f[cur][j])
                        f[i][j] = f[i][cur] + f[cur][j];
            cur++;
        }
        if (t[x] > d || t[y] > d || f[x][y] == INF)
            cout << -1 << endl;
        else
            cout << f[x][y] << endl;
    }
    return 0;
}
```

## --------------------------------------------------------------------------
### P1396 营救
## --------------------------------------------------------------------------

题目: 城市拥挤度为边的权值，求从起点s到终点t的路径中，经过的最大拥挤度的最小值。

思路: 经典的最小化最大边权问题。方法一：二分答案+BFS判定；方法二：最小生成树Kruskal。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

struct Edge { int u, v, w; } e[20005];
int fa[10005];

int find(int x) { return fa[x] == x ? x : fa[x] = find(fa[x]); }
bool cmp(Edge a, Edge b) { return a.w < b.w; }

int main() {
    int n, m, s, t;
    cin >> n >> m >> s >> t;
    for (int i = 1; i <= n; i++) fa[i] = i;
    for (int i = 0; i < m; i++)
        cin >> e[i].u >> e[i].v >> e[i].w;
    sort(e, e + m, cmp);
    for (int i = 0; i < m; i++) {
        fa[find(e[i].u)] = find(e[i].v);
        if (find(s) == find(t)) {
            cout << e[i].w << endl;
            return 0;
        }
    }
    return 0;
}
```

> > #include <iostream>
#include <deque>

vector<pair<int,int>> g[100005];
int main(){
    int n,m,s,t;
    cin>>n>>m>>s>>t;
    for(int i=1;i<=n;i++) dist[i]=INF;
    for(int i=0;i<m;i++){int u,v,w;cin>>u>>v>>w;g[u].push_back({v,w});g[v].push_back({u,w});}
    deque<int> q;
    dist[s]=0; q.push_front(s);
    while(!q.empty()){
        int u=q.front();q.pop_front();
        for(auto [v,w]:g[u]){
            if(dist[v]>dist[u]+w){
                dist[v]=dist[u]+w;
                if(w==0) q.push_front(v); else q.push_back(v);

## ==========================================================================
C++ 算法技巧 — 暴力枚举 (Brute Force Enumeration)
## ==========================================================================

暴力枚举即通过遍历所有可能的情况来寻找答案，是解决问题的最直接方法。

**前置知识**: [[顺序]]、[[分支]]、[[循环]]、[[数组]]

## --------------------------------------------------------------------------
### P1008 [NOIP1998 普及组] 三连击
## --------------------------------------------------------------------------

题目: 将1~9分成三组，每组三个数字组成三位数，且这三个三位数构成1:2:3的比例。

思路: 枚举第一个三位数（123~329），计算第二个和第三个，判断9个数字是否恰好为1~9。

```cpp
#include <iostream>
using namespace std;

bool check(int a, int b, int c) {
    int cnt[10] = {0};
    cnt[a / 100]++; cnt[a / 10 % 10]++; cnt[a % 10]++;
    cnt[b / 100]++; cnt[b / 10 % 10]++; cnt[b % 10]++;
    cnt[c / 100]++; cnt[c / 10 % 10]++; cnt[c % 10]++;
    for (int i = 1; i <= 9; i++)
        if (cnt[i] != 1) return false;
    return true;
}

int main() {
    for (int a = 123; a <= 329; a++) {
        int b = a * 2, c = a * 3;
        if (check(a, b, c))
            cout << a << " " << b << " " << c << endl;
    }
    return 0;
}
```

## --------------------------------------------------------------------------
### P1149 [NOIP2008 提高组] 火柴棒等式
## --------------------------------------------------------------------------

题目: 用n根火柴棒拼出形如A+B=C的等式，其中A、B、C是由火柴棒拼出的整数（0~9各有固定根数）。
问一共有多少种不同的等式。

思路: 先打表每个数字需要的火柴棒数，然后枚举A和B，计算C=A+B判断总火柴数是否恰好用完。

```cpp
#include <iostream>
using namespace std;

int cost[10] = {6, 2, 5, 5, 4, 5, 6, 3, 7, 6};

int need(int x) {
    if (x == 0) return cost[0];
    int sum = 0;
    while (x) { sum += cost[x % 10]; x /= 10; }
    return sum;
}

int main() {
    int n, ans = 0;
    cin >> n;
    n -= 4; // 减去加号和等号的4根
    for (int a = 0; a <= 999; a++)
        for (int b = 0; b <= 999; b++)
            if (need(a) + need(b) + need(a + b) == n)
                ans++;
    cout << ans << endl;
    return 0;
}
```

### P1217 [USACO1.5] 回文质数 Prime Palindromes
## --------------------------------------------------------------------------

题目: 找出[a,b]区间内所有回文质数。

思路: 先枚举生成所有回文数（比枚举所有数更高效），再判断是否为质数。

```cpp
#include <iostream>
using namespace std;

bool isPrime(int x) {
    if (x < 2) return false;
    for (int i = 2; i * i <= x; i++)
        if (x % i == 0) return false;
    return true;
}

int main() {
    int a, b;
    cin >> a >> b;
    // 生成4位回文数
    for (int d1 = 1; d1 <= 9; d1 += 2)
        for (int d2 = 0; d2 <= 9; d2++) {
            int num = d1 * 1000 + d2 * 100 + d2 * 10 + d1;
            if (num >= a && num <= b && isPrime(num))
                cout << num << endl;
        }
    // 生成5位回文数
    for (int d1 = 1; d1 <= 9; d1 += 2)
        for (int d2 = 0; d2 <= 9; d2++)
            for (int d3 = 0; d3 <= 9; d3++) {
                int num = d1 * 10000 + d2 * 1000 + d3 * 100 + d2 * 10 + d1;
                if (num >= a && num <= b && isPrime(num))
                    cout << num << endl;
            }
    // 同理可生成3、6、7位回文数（实际题目a,b ≤ 100,000,000）
    return 0;
}
```

### P1618 三连击（升级版）
## --------------------------------------------------------------------------

题目: 将1~9分成三组，每组三个数字组成三位数，且这三个三位数构成A:B:C的比例。
给定A、B、C，输出所有符合条件的三位数组合。

思路: 与P1008类似，枚举第一个数，按比例计算第二个和第三个，检查九个数字是否恰好为1~9。

```cpp
#include <iostream>
using namespace std;

bool check(int a, int b, int c) {
    int cnt[10] = {0};
    cnt[a / 100]++; cnt[a / 10 % 10]++; cnt[a % 10]++;
    cnt[b / 100]++; cnt[b / 10 % 10]++; cnt[b % 10]++;
    cnt[c / 100]++; cnt[c / 10 % 10]++; cnt[c % 10]++;
    for (int i = 1; i <= 9; i++)
        if (cnt[i] != 1) return false;
    return true;
}

int main() {
    int A, B, C;
    bool found = false;
    cin >> A >> B >> C;
    for (int x = 123; x <= 987; x++) {
        if (x * B % A || x * C % A) continue;
        int y = x * B / A, z = x * C / A;
        if (z > 987) break;
        if (check(x, y, z)) {
            cout << x << " " << y << " " << z << endl;
            found = true;
        }
    }
    if (!found) cout << "No!!!" << endl;
    return 0;
}
```

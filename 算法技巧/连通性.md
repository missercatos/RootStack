## --------------------------------------------------------------------------
### P3367 [模板] 并查集
## --------------------------------------------------------------------------

题目: 支持合并集合和查询两个元素是否在同一集合的操作。

思路: 使用并查集，路径压缩+按秩合并优化。

```cpp
#include <iostream>
using namespace std;
int fa[10005], rk[10005];

int find(int x) {
    return fa[x] == x ? x : fa[x] = find(fa[x]);
}

void merge(int x, int y) {
    x = find(x), y = find(y);
    if (x == y) return;
    if (rk[x] < rk[y]) swap(x, y);
    fa[y] = x;
    if (rk[x] == rk[y]) rk[x]++;
}

int main() {
    int n, m;
    cin >> n >> m;
    for (int i = 1; i <= n; i++) fa[i] = i;
    while (m--) {
        int op, x, y;
        cin >> op >> x >> y;
        if (op == 1) merge(x, y);
        else cout << (find(x) == find(y) ? "Y" : "N") << endl;
    }
    return 0;
}
```

## --------------------------------------------------------------------------
### P3385 [模板] 负环
## --------------------------------------------------------------------------

题目: 判断有向图中是否存在负环。

思路: 使用SPFA算法，记录每个顶点的入队次数，大于n次则存在负环。

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int INF = 1e9;
int n, m, dist[2005], cnt[2005];
bool inq[2005];
vector<pair<int,int>> g[2005];

bool spfa() {
    queue<int> q;
    for (int i = 1; i <= n; i++) {
        dist[i] = 0, cnt[i] = 0, inq[i] = true;
        q.push(i);
    }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) {
                    q.push(v);
                    inq[v] = true;
                    if (++cnt[v] > n) return true;
                }
            }
        }
    }
    return false;
}

int main() {
    int T;
    cin >> T;
    while (T--) {
        cin >> n >> m;
        for (int i = 1; i <= n; i++) g[i].clear();
        for (int i = 0, u, v, w; i < m; i++) {
            cin >> u >> v >> w;
            g[u].push_back({v, w});
            if (w >= 0) g[v].push_back({u, w});
        }
        cout << (spfa() ? "YES" : "NO") << endl;
    }
    return 0;
}
```

## --------------------------------------------------------------------------
### P5960 [模板] 差分约束算法
## --------------------------------------------------------------------------

题目: 给出m个不等式形如 x_a - x_b ≤ c，求一组合法的解。

思路: 差分约束系统转化为最短路径问题：不等式 x_a - x_b ≤ c 等价于 x_a ≤ x_b + c，
可以视为从b到a有一条权值为c的边。跑SPFA求最长路判断是否可行（有负环则无解）。

```cpp
#include <iostream>
#include <vector>
#include <queue>
using namespace std;

const int INF = 1e9;
int n, m, dist[5005], cnt[5005];
bool inq[5005];
vector<pair<int,int>> g[5005];

bool spfa() {
    queue<int> q;
    for (int i = 1; i <= n + 1; i++) {
        dist[i] = 0; cnt[i] = 0;
        inq[i] = true;
        q.push(i);
    }
    while (!q.empty()) {
        int u = q.front(); q.pop();
        inq[u] = false;
        for (auto [v, w] : g[u]) {
            if (dist[v] > dist[u] + w) {
                dist[v] = dist[u] + w;
                if (!inq[v]) {
                    q.push(v); inq[v] = true;
                    if (++cnt[v] > n + 1) return false;
                }
            }
        }
    }
    return true;
}

int main() {
    cin >> n >> m;
    for (int i = 0, u, v, w; i < m; i++) {
        cin >> u >> v >> w;
        g[v].push_back({u, w}); // x_u <= x_v + w
    }
    if (!spfa()) cout << "NO" << endl;
    else {
        for (int i = 1; i <= n; i++)
            cout << dist[i] << " ";
        cout << endl;
    }
    return 0;
}
```

## --------------------------------------------------------------------------
### P3387 [模板] 缩点
## --------------------------------------------------------------------------

题目: 给定有向图，可以将一个强连通分量缩成一个点。缩点后的图是DAG，求从任意点出发的最大路径权值和。

思路: Tarjan求强连通分量(SCC)，缩点后在新DAG上做DP（拓扑序或记忆化搜索）。

```cpp
#include <iostream>
#include <vector>
#include <stack>
using namespace std;

int n, m, a[10005], dfn[10005], low[10005], scc[10005], w[10005], dp[10005];
int timer, scc_cnt;
bool inStack[10005];
stack<int> st;
vector<int> g[10005], ng[10005];

void tarjan(int u) {
    dfn[u] = low[u] = ++timer;
    st.push(u); inStack[u] = true;
    for (int v : g[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else if (inStack[v]) {
            low[u] = min(low[u], dfn[v]);
        }
    }
    if (dfn[u] == low[u]) {
        scc_cnt++;
        while (true) {
            int v = st.top(); st.pop();
            inStack[v] = false;
            scc[v] = scc_cnt;
            w[scc_cnt] += a[v];
            if (v == u) break;
        }
    }
}

int dfs(int u) {
    if (dp[u]) return dp[u];
    dp[u] = w[u];
    int mx = 0;
    for (int v : ng[u])
        mx = max(mx, dfs(v));
    dp[u] += mx;
    return dp[u];
}

int main() {
    cin >> n >> m;
    for (int i = 1; i <= n; i++) cin >> a[i];
    for (int i = 0, u, v; i < m; i++) {
        cin >> u >> v;
        g[u].push_back(v);
    }
    for (int i = 1; i <= n; i++)
        if (!dfn[i]) tarjan(i);
    for (int u = 1; u <= n; u++)
        for (int v : g[u])
            if (scc[u] != scc[v])
                ng[scc[u]].push_back(scc[v]);
    int ans = 0;
    for (int i = 1; i <= scc_cnt; i++)
        ans = max(ans, dfs(i));
    cout << ans << endl;
    return 0;
}
```

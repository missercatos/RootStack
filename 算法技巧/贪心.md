## ==========================================================================
C++ 算法技巧 — 贪心 (Greedy)
## ==========================================================================

贪心算法在每一步选择中都采取当前状态下最优的决策，从而希望最终结果也是最优的。

**前置知识**: [[数组]]、[[排序]]、[[循环]]

## --------------------------------------------------------------------------
### P1223 排队接水
## --------------------------------------------------------------------------

题目: n个人排队打水，每人需要T[i]时间。求一种排队顺序使得所有人的平均等待时间最小。

思路: 短作业优先——按打水时间升序排列，等待时间为前面所有人打水时间之和。

```cpp
#include <iostream>
#include <algorithm>
#include <cstdio>
using namespace std;

struct P { int t, id; };
bool cmp(P a, P b) { return a.t < b.t; }

int main() {
    int n;
    P p[1005];
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> p[i].t;
        p[i].id = i + 1;
    }
    sort(p, p + n, cmp);
    double total = 0, cur = 0;
    for (int i = 0; i < n; i++) {
        cout << p[i].id << " ";
        total += cur;
        cur += p[i].t;
    }
    printf("\n%.2f\n", total / n);
    return 0;
}
```

## --------------------------------------------------------------------------
### P1090 [NOIP2004 提高组] 合并果子
## --------------------------------------------------------------------------

题目: n堆果子每堆重量a_i，每次合并两堆消耗体力等于两堆重量之和，求合并成一堆的最小体力消耗。

思路: 每次合并最小的两堆（哈夫曼树思想），用优先队列（小根堆）维护。

```cpp
#include <iostream>
#include <queue>
using namespace std;

int main() {
    int n, ans = 0;
    priority_queue<int, vector<int>, greater<int>> pq;
    cin >> n;
    for (int i = 0, x; i < n; i++) {
        cin >> x;
        pq.push(x);
    }
    while (pq.size() > 1) {
        int a = pq.top(); pq.pop();
        int b = pq.top(); pq.pop();
        int sum = a + b;
        ans += sum;
        pq.push(sum);
    }
    cout << ans << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P1106 删数问题
## --------------------------------------------------------------------------

题目: 输入一个高精度正整数n，去掉其中s个数字后剩下的数字按原顺序组成新数，求新数的最小值。

思路: 贪心策略: 每次从左到右找第一个比右边大的数字删除（这个数字的"贡献"太大），重复s次。

```cpp
#include <iostream>
#include <string>
using namespace std;

int main() {
    string n;
    int s;
    cin >> n >> s;
    while (s--) {
        int i = 0;
        while (i < (int)n.size() - 1 && n[i] <= n[i + 1]) i++;
        n.erase(i, 1);
    }
    while (n.size() > 1 && n[0] == '0') n.erase(0, 1);
    cout << n << endl;
    return 0;
}
```

### P2240 部分背包问题
## --------------------------------------------------------------------------

题目: 体积为M的背包和N件物品，每件物品有体积v_i和价值w_i，物品可任意切割（取体积的一部分则按比例获得价值）。
求背包能装的最大总价值。

思路: 按性价比（价值/体积）降序排列，贪心装入，最后一件可能切割。

```cpp
#include <iostream>
#include <algorithm>
#include <cstdio>
using namespace std;

struct Item { double w, v, ratio; };
bool cmp(Item a, Item b) { return a.ratio > b.ratio; }

int main() {
    int n; double m;
    Item a[105];
    cin >> n >> m;
    for (int i = 0; i < n; i++) {
        cin >> a[i].w >> a[i].v;
        a[i].ratio = a[i].v / a[i].w;
    }
    sort(a, a + n, cmp);
    double ans = 0;
    for (int i = 0; i < n && m > 1e-9; i++) {
        double take = min(m, a[i].w);
        ans += take * a[i].ratio;
        m -= take;
    }
    printf("%.2f\n", ans);
    return 0;
}
```

#### 动手练习题

> [!question] 动手练习 1
> **题目**: n个活动各有开始和结束时间，选出最多不冲突的活动数。
> 
> **提示**: 按结束时间排序，贪心选择不冲突的活动。
> 
> > [!success]- 点击查看参考答案
> > ```cpp
> > #include <iostream>
#include <algorithm>
using namespace std;
struct Act { int s, e; };
bool cmp(Act a, Act b) { return a.e < b.e; }
int main() {
    int n;
    cin >> n;
    Act a[1005];
    for (int i = 0; i < n; i++) cin >> a[i].s >> a[i].e;
    sort(a, a+n, cmp);
    int cnt = 1, last = a[0].e;
    for (int i = 1; i < n; i++)
        if (a[i].s >= last) { cnt++; last = a[i].e; }
    cout << cnt << endl;
    return 0;
}
> > ```

## --------------------------------------------------------------------------
### P1601 A+B Problem（高精）
## --------------------------------------------------------------------------

题目: 高精度加法，求两个不超过10^500位的非负整数之和。

思路: 将数字按位存入数组，模拟竖式加法，从低位向高位逐位相加并处理进位。

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string a, b;
    cin >> a >> b;
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    int len = max(a.size(), b.size());
    int sum[505] = {0};
    for (int i = 0; i < len; i++) {
        if (i < (int)a.size()) sum[i] += a[i] - '0';
        if (i < (int)b.size()) sum[i] += b[i] - '0';
    }
    for (int i = 0; i < len; i++) {
        if (sum[i] >= 10) {
            sum[i + 1] += sum[i] / 10;
            sum[i] %= 10;
        }
    }
    if (sum[len] > 0) len++;
    for (int i = len - 1; i >= 0; i--)
        cout << sum[i];
    cout << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P2142 高精度减法
## --------------------------------------------------------------------------

题目: 高精度减法，求两个正整数之差（保证a≥b）。

思路: 模拟竖式减法，不够向前一位借1。

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string a, b;
    cin >> a >> b;
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    int len = a.size(), res[10100] = {0};
    for (int i = 0; i < len; i++) {
        int x = a[i] - '0';
        int y = (i < (int)b.size()) ? b[i] - '0' : 0;
        if (x < y) {
            a[i + 1]--;
            x += 10;
        }
        res[i] = x - y;
    }
    while (len > 1 && res[len - 1] == 0) len--;
    for (int i = len - 1; i >= 0; i--)
        cout << res[i];
    cout << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P1303 A*B Problem（高精）
## --------------------------------------------------------------------------

题目: 高精度乘法，求两个不超过2000位的非负整数之积。

思路: 双重循环模拟竖式乘法，第i位和第j位相乘的结果贡献到第i+j位。

```cpp
#include <iostream>
#include <string>
#include <algorithm>
using namespace std;

int main() {
    string a, b;
    cin >> a >> b;
    reverse(a.begin(), a.end());
    reverse(b.begin(), b.end());
    int res[4005] = {0};
    for (int i = 0; i < (int)a.size(); i++)
        for (int j = 0; j < (int)b.size(); j++)
            res[i + j] += (a[i] - '0') * (b[j] - '0');
    int len = a.size() + b.size();
    for (int i = 0; i < len; i++) {
        res[i + 1] += res[i] / 10;
        res[i] %= 10;
    }
    while (len > 1 && res[len - 1] == 0) len--;
    for (int i = len - 1; i >= 0; i--)
        cout << res[i];
    cout << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P1009 [NOIP1998 普及组] 阶乘之和
## --------------------------------------------------------------------------

题目: 用高精度计算S=1!+2!+3!+...+n!，其中n≤50。

思路: 高精度乘法和加法结合：用一个数组表示阶乘值，逐项累加。

```cpp
#include <iostream>
using namespace std;

int res[100] = {0}, fac[100] = {1};

int main() {
    int n;
    cin >> n;
    for (int i = 1; i <= n; i++) {
        for (int j = 0; j < 100; j++)
            fac[j] *= i;
        for (int j = 0; j < 99; j++) {
            fac[j + 1] += fac[j] / 10;
            fac[j] %= 10;
        }
        for (int j = 0; j < 100; j++) {
            res[j] += fac[j];
            res[j + 1] += res[j] / 10;
            res[j] %= 10;
        }
    }
    int p = 99;
    while (p > 0 && res[p] == 0) p--;
    for (; p >= 0; p--) cout << res[p];
    cout << endl;
    return 0;
}
```

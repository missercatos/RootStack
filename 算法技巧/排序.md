C++ 算法技巧 — 排序 (Sorting)
## ==========================================================================

排序是将无序数据按一定规则排列的算法，是许多高级算法的基础。

**前置知识**: [[数组]]、[[循环]]、[[函数结构体]]

## --------------------------------------------------------------------------
### P1177 快速排序
## --------------------------------------------------------------------------

题目: 读入n个数字并排序后输出。

思路: 使用标准库sort或手写快速排序。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int a[100005];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    for (int i = 0; i < n; i++) cout << a[i] << " ";
    return 0;
}
```

## --------------------------------------------------------------------------
### P1059 [NOIP2006 普及组] 明明的随机数
## --------------------------------------------------------------------------

题目: N个随机整数，要求去重后从小到大排序输出。

思路: 使用sort排序后用vector去重，或直接用set容器。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;
int a[1005];

int main() {
    int n;
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    sort(a, a + n);
    int cnt = unique(a, a + n) - a;
    cout << cnt << endl;
    for (int i = 0; i < cnt; i++) cout << a[i] << " ";
    return 0;
}
```

## --------------------------------------------------------------------------
### P1093 [NOIP2007 普及组] 奖学金
## --------------------------------------------------------------------------

题目: N个学生的语文、数学、英语成绩，按总分降序排列；总分相同按语文降序；语文也相同按学号升序。

思路: 使用结构体+自定义排序规则。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

struct Stu {
    int id, ch, ma, en, sum;
};

bool cmp(Stu a, Stu b) {
    if (a.sum != b.sum) return a.sum > b.sum;
    if (a.ch != b.ch) return a.ch > b.ch;
    return a.id < b.id;
}

int main() {
    int n;
    Stu s[305];
    cin >> n;
    for (int i = 0; i < n; i++) {
        cin >> s[i].ch >> s[i].ma >> s[i].en;
        s[i].id = i + 1;
        s[i].sum = s[i].ch + s[i].ma + s[i].en;
    }
    sort(s, s + n, cmp);
    for (int i = 0; i < 5 && i < n; i++)
        cout << s[i].id << " " << s[i].sum << endl;
    return 0;
}
```

## --------------------------------------------------------------------------
### P1068 [NOIP2009 普及组] 分数线划定
## --------------------------------------------------------------------------

题目: N个选手，计划录取人数m的150%（向下取整），按分数降序排序，
录取分数线为计划录取的最后一名选手的分数。

思路: 排序后取划分点，同时考虑同分并列的情况。

```cpp
#include <iostream>
#include <algorithm>
using namespace std;

struct P { int id, score; };
bool cmp(P a, P b) {
    if (a.score != b.score) return a.score > b.score;
    return a.id < b.id;
}

int main() {
    int n, m;
    P p[5005];
    cin >> n >> m;
    m = m * 1.5;
    for (int i = 0; i < n; i++)
        cin >> p[i].id >> p[i].score;
    sort(p, p + n, cmp);
    int line = p[m - 1].score;
    int cnt = m;
    while (cnt < n && p[cnt].score == line) cnt++;
    cout << line << " " << cnt << endl;
    for (int i = 0; i < cnt; i++)
        cout << p[i].id << " " << p[i].score << endl;
    return 0;
}
```

> [!question] 动手练习 3
> **题目**: 实现简单的冒泡排序，输入n个整数，输出排序结果。
> 
> **提示**: 双重循环，相邻元素比较交换
> 
> > [!success]- 点击查看参考答案
> > ```cpp
> > #include <iostream>
using namespace std;
int main() {
    int n, a[1005];
    cin >> n;
    for (int i = 0; i < n; i++) cin >> a[i];
    for (int i = 0; i < n-1; i++)
        for (int j = 0; j < n-1-i; j++)
            if (a[j] > a[j+1]) swap(a[j], a[j+1]);
    for (int i = 0; i < n; i++) cout << a[i] << " ";
    return 0;
}
> > ```
